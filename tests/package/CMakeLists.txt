# An outside consumer of the installed package: the existing C++ smoke test, built
# against find_package(iya_wsfed) rather than this source tree. The Docker test stage
# builds and runs it against both the static and the shared install.
cmake_minimum_required(VERSION 3.16)
project(iya_wsfed_package_test LANGUAGES C CXX)
find_package(iya_wsfed ${iya_wsfed_EXPECTED_VERSION} EXACT REQUIRED)
add_executable(consumer ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/cpp_smoke.cpp)
target_link_libraries(consumer PRIVATE iya::wsfed)
